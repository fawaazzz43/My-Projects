DIR ?= monitored_dir
MALICIOUS_DIR ?= quarantine
INTERVAL ?= 5

all: setup

setup:
	@mkdir -p $(MALICIOUS_DIR)[cite: 4]
	@mkdir -p $(DIR)

run-antivirus: setup
	bash antivirusd.sh $(DIR) $(MALICIOUS_DIR) $(INTERVAL)[cite: 4]

run-restore: setup
	bash restore.sh $(DIR) $(MALICIOUS_DIR)[cite: 4]

clean:
	rm -f directory-info.last directory-info.new whitelist.txt
